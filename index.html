Koneksi Google Spreadsheet Backend

Hubungkan aplikasi ini secara langsung dengan akun Google Spreadsheet milik Anda menggunakan Apps Script Web App.

Tempelkan URL Web App yang didapatkan dari Google Apps Script setelah dipublikasikan.

Panduan Membuat Google Apps Script (Hanya 1x Setup)

  1. Buka Google Sheets baru atau yang sudah ada di akun Google Anda.
  2. Di menu atas Google Sheets, klik Ekstensi (Extensions) → Apps Script.
  3. Hapus semua kode bawaan di editor Apps Script, lalu Copy & Paste kode script dibawah ini:
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Tambahkan Header jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Waktu Submit", "Tanggal Penilaian", "Hari", "Lokasi", "Nama Pelapor", 
        "Nama Karyawan Dinilai", "Section", "Skor Pencapaian Kerja", "Skor Sikap", 
        "Skor Kepemimpinan", "Skor Harmonisasi", "Total Skor", "Grade", "Catatan"
      ]);
    }
    
    // Tambahkan Baris Data Penilaian
    sheet.appendRow([
      new Date(),
      data.date,
      data.day,
      data.location,
      data.reporter,
      data.targetMember,
      data.section,
      data.scorePencapaian,
      data.scoreSikap,
      data.scoreKepemimpinan,
      data.scoreHarmonisasi,
      data.totalScore,
      data.grade,
      data.notes
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ "result": "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ "result": "error", "error": err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
